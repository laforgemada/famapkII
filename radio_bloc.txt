    function formatTime(totalSeconds) {

        totalSeconds = Math.max(
            0,
            Math.floor(totalSeconds)
        );

        const hours = Math.floor(totalSeconds / 3600);

        const minutes = Math.floor(
            (totalSeconds % 3600) / 60
        );

        const seconds = totalSeconds % 60;

        return [
            String(hours).padStart(2, '0'),
            String(minutes).padStart(2, '0'),
            String(seconds).padStart(2, '0')
        ].join(':');
    }

    async function syncListeningTime() {

        if (!isAuthenticated()) {

            console.log(
                '[Radio] InvitÃ© : aucun temps envoyÃ© au serveur.'
            );

            return;
        }

        if (isSyncing) {
            return;
        }

        const now = Date.now();

        if (!lastTick) {
            lastTick = now;
            return;
        }

        const elapsedSeconds =
            Math.floor((now - lastTick) / 1000);

        if (elapsedSeconds <= 0) {
            return;
        }

        /*
         * On limite volontairement ce qui est envoyÃ©.
         * Le backend possÃ¨de Ã©galement sa propre protection.
         */
        const secondsToSend =
            Math.min(elapsedSeconds, 120);

        lastTick = now;

        isSyncing = true;

        try {

            const token = getToken();

            const response = await fetch(
                `${API_BASE_URL}/api/radio/listen`,
                {
                    method: 'POST',

                    headers: {
                        'Content-Type': 'application/json',

                        'Authorization':
                            `Bearer ${token}`
                    },

                    body: JSON.stringify({
                        seconds: secondsToSend
                    })
                }
            );

            const data = await response.json();

            console.log(
                '[Radio] Synchronisation :',
                data
            );

            if (!response.ok) {

                console.warn(
                    '[Radio] Erreur serveur :',
                    data
                );

                return;
            }

            /*
             * Le serveur est la source de vÃ©ritÃ©.
             */
            if (
                typeof data.radioListenSeconds ===
                'number'
            ) {

                accumulatedSeconds =
                    data.radioListenSeconds;
            }

            updateDisplay();

            /*
             * Si une rÃ©compense vient d'Ãªtre obtenue,
             * on informe l'utilisateur.
             */
            if (
                data.pointsEarned &&
                Number(data.pointsEarned) > 0
            ) {

                showRewardNotification(
                    Number(data.pointsEarned)
                );
            }

        } catch (error) {

            console.error(
                '[Radio] Erreur synchronisation :',
                error
            );

            /*
             * En cas d'erreur rÃ©seau, on ne perd pas
             * le temps affichÃ© localement.
             */
            accumulatedSeconds += secondsToSend;

            updateDisplay();

        } finally {

            isSyncing = false;
        }
    }
